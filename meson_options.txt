# Trace
option('trace',     type: 'boolean', value: true,  description: 'Enable tracer')
option('itrace',    type: 'boolean', value: true,  description: 'Enable instruction tracer')

option('ftrace_stack_max', type: 'integer', value: 64, description: 'Max function call stack depth')

option('mtrace',    type: 'boolean', value: true,  description: 'Enable memory tracer')
option('mtrace_cond', type: 'string', value: 'true', description: 'Mtrace condition expression')
option('etrace',    type: 'boolean', value: true,  description: 'Enable exception tracer')
option('etrace_buf_size', type: 'integer', value: 16, description: 'Exception trace buffer size')
